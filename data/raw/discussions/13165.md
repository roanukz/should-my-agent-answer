## Accepted answer by YuriiMotov (MEMBER)

This is because `partial` with `async` functions isn't supported in `Depends`: issue https://github.com/fastapi/fastapi/issues/10720

You can workaround it by rewriting `get_profile_load_options` like this:
```python
def get_profile_load_options(*load_options) -> Callable:
    print(f"UTILS get_profile_load_options {load_options=}")

    async def parametrized(
        db: AsyncSession = Depends(get_db),
        profile_id: int | None = None,
    ) -> Profile:
        return await get_current_profile(
            db=db, profile_id=profile_id, load_options=list(load_options)
        )
    return parametrized
```
Or this:
```python
def get_profile_load_options(*load_options):
    print(f"UTILS get_profile_load_options {load_options=}")

    async def get_current_profile(
        db: AsyncSession = Depends(get_db),
        profile_id: int | None = None,
    ) -> Profile:
        nonlocal load_options
        if not load_options:
            load_options = []

        profile = await db.scalar(
            select(Profile).filter(Profile.id == profile_id).options(*load_options)
        )
        if not profile:
            raise HTTPException(status_code=404, detail="Profile not found")
        return profile

    return get_current_profile
```
