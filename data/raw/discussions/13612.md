### Example Code

```python
@users.post("/login")
async def login(request:model_request_login,language=Header(None)):
    if request.captcha == None:
        return response("captcha_empty")
    if  await chack_captcha(request.captcha):
        result  =  await processing_login(request)
        if result == 200:
            token = await generate_token(request.username)
            data = {"token":token}
            return response(message="successful_login",data=data,lang=language)
        elif result == 403:
            return response(message="login_failed")
        elif result == 404:
            return response(message="not_create_user",lang=language)
    else:
        return response(message="invalid_captcha")
```
