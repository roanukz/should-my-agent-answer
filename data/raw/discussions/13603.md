## Question

The user just uses the endpoint
> /export?format= …


This is the sample code for JSON format:


```
from fastapi import FastAPI, Query

import pandas as pd

# Sample data (connection with the dbase instead…)

	{"id": 1, "name": "Sebastián", "age": 30},
	{"id": 2, "name": "George", "age": 35},
 	{"id": 3, "name": "Joanna", "age": 22},
]



@app.get("/export")


async def export_data(format: str = Query("json", enum=["json", "... other formats to be"])):
	df = pd.DataFrame(data)

	if format == "json":
    	return JSONResponse(content=df.to_dict(orient="records"))


	elif format == "other formats to be...":
# Here will be new format to be…




	return JSONResponse(content={"error": "Invalid format"}, status_code=400)
```
