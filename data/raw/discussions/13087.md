```python
@thyroid_router.post("/predict", response_model=MILResponseModel)
async def predict(
    query: RequestModel,
    mil_predictor: MILPredictor = Depends(get_mil_predictor),
    config=Depends(get_config),
    logger: Logger = Depends(get_logger),
) -> MILResponseModel:


    logger.info(
        f"Accept request slide_path({query.slide_path}), Thread ID({threading.get_ident()})"
    )

    deepzoom_level = get_deepzoom_level(config["mpp"], OpenSlide(query.slide_path))
    tiler = Tiler(
        config.tile_size,
        config.overlap,
        config.limit_bounds,
        deepzoom_level=deepzoom_level,
        n_workers=config.tile_workers,
        patch_filter=PatchFilter(
            [
                check_small_compressed_format,
            ]
        ),
        logger=logger,
    )
    output_queue: queue.Queue = tiler.do_tile_otsu(query.slide_path)
    tiler.join()
    patches = Patches.from_queue(output_queue)
    logger.info(f"End of tiling N={len(patches)} ")

    logger.debug("Start of slide_level prediction")
    confidences, attention_weights = mil_predictor.predict(patches)
    logger.debug("End of prediction")

    # slide-evel prediction
    logger.debug("Confidence to float")
    confidences = confidences.ravel()
    max_idx = confidences.ravel().argmax()
    slide_level_label: str = BinaryLabels(max_idx).name
    slide_confidence = float(confidences[max_idx])

    logger.debug("Build response model")
    slide = OpenSlide(query.slide_path)
    deepzoom_level: int = get_deepzoom_level(config.mpp, slide)
    slide_result = MILSlideResult(
        confidence=slide_confidence,
        label=slide_level_label,
    )
    response = MILResponseModel.build(
        query=query,
        slide_result=slide_result,
        patches=patches,
        heatmap_scores=attention_weights,
        patch_mpp=config.mpp,
        model_uuid=config.model_uuid,
        model_name=mil_predictor.mil_model.__class__.__name__,
        top_patch_percent=config.top_patch_percent,
    )
    logger.info(
        f"End of request slide_path({query.slide_path}), Thread ID({threading.get_ident()})"
    )

    return response
```
