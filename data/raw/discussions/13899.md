### Description

Hi all!

As you know to implement multiple authorization schemes for a single endpoint one need to something like this:

```python
from fastapi import FastAPI, Depends, HTTPException
from fastapi.security import HTTPBasicCredentials, HTTPBasic, HTTPAuthorizationCredentials, HTTPBearer

app = FastAPI()

def multiple_auth_schemes(
    auth_1_res: HTTPBasicCredentials = Depends(HTTPBasic()),
    auth_2_res: HTTPAuthorizationCredentials = Depends(HTTPBearer()),
):
    for auth_res in (auth_1_res, auth_2_res):
        if auth_res is not None:
            return auth_res
    raise HTTPException(status_code=401)


@app.get("/test", dependencies=[Depends(multiple_auth_schemes)])
async def test_multiple_auth_schemes():
    pass
```

## Accepted answer by albertalexandrov (NONE)

It seems I did it :)

```python
from inspect import Signature, Parameter
from inspect import Signature, Parameter
from typing import Annotated, Any

from fastapi import FastAPI, Depends, HTTPException
from fastapi.security import HTTPBasic, HTTPBearer
from starlette.requests import Request

app = FastAPI()


def dynamic_multiple_auth_schemes(*authenticators):
    
    def wrapper(request: Request, **kwargs):
        print(request, kwargs)
        for auth_res in kwargs.values():
            if auth_res is not None:
                return auth_res
        raise HTTPException(status_code=401)
    
    parameters = [
        Parameter("request", Parameter.POSITIONAL_OR_KEYWORD, annotation=Request),
    ]
    for idx, authenticator in enumerate(authenticators, 1):
        name = f"param{idx}"
        parameter = Parameter(
            name,
            Parameter.POSITIONAL_OR_KEYWORD,
            annotation=Annotated[Any, Depends(authenticator)],
        )
        parameters.append(parameter)
    wrapper.__signature__ = Signature(parameters)
    return wrapper


@app.get(
    "/test",
    dependencies=[
        Depends(dynamic_multiple_auth_schemes(HTTPBasic(auto_error=False), HTTPBearer(auto_error=False)))
    ]
)
async def test_multiple_auth_schemes():
    pass
```
