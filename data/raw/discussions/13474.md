### Example Code

```python
from fastapi import APIRouter, Depends
from sqlmodel import Session, SQLModel, Field, select
from datetime import datetime

documentos_router = APIRouter()

class DocumentoElectronico(SQLModel, table=True):
    __tablename__ = "RC_V_ComprobantesElectronicos"
    id_pro: int = Field(primary_key=True)
    id_emi: int = Field()
    id_rec: int = Field()
    tip_doc: str = Field(max_length=10)
    tip_doc_nom: str = Field(max_length=255)
    num_doc: str = Field(max_length=50)
    doc_sus: str = Field(max_length=50, nullable=True)
    data_xml: bytes = Field(nullable=True)
    data_pdf: bytes = Field(nullable=True)
    fec_doc: datetime = Field()

@documentos_router.get("/documentos")
def get_documentos(
    skip: int = 0,
    limit: int = 10,
    fechaDesde: str = None,
    fechaHasta: str = None,
    tipoDocumento: str = None,
    id_emi: int = None,
    id_rec: int = None,
    session: Session = Depends(get_db),
):
    query = select(DocumentoElectronico)

    if id_emi:
        query = query.where(DocumentoElectronico.id_emi == id_emi)

    if id_rec:
        query = query.where(DocumentoElectronico.id_rec == id_rec)

    if tipoDocumento:
        query = query.where(DocumentoElectronico.tip_doc == tipoDocumento)

    if fechaDesde and fechaHasta:
        query = query.where(
            DocumentoElectronico.fec_doc >= datetime.strptime(fechaDesde, "%Y-%m-%d"),
            DocumentoElectronico.fec_doc <= datetime.strptime(fechaHasta, "%Y-%m-%d"),
        )

    documentos = session.exec(
        query.order_by(DocumentoElectronico.fec_doc.desc()).offset(skip).limit(limit)
    ).all()

    return documentos
```


### Description

I am working with a database in which I have a view called RC_V_ComprobantesElectronicos, but when I try to access this view from FastAPI, I find that it cannot locate it. Even though the view exists in the database, when I try to query it with SQLModel, FastAPI does not recognize it as a valid table.

## Accepted answer by DScabezas (NONE)

I made a change and it worked, although it requires more steps than I wanted. I already had an interface with the same elements, so now there are additional steps and more code, but at least it's functional.

```python
@documentos_router.get("/documento", response_model=List[DocumentoResponse])
def get_documento(
    filtros: DocumentoFilter = Depends(),
    session: Session = Depends(get_db),
    current_user=Depends(get_current_user),
) -> List[DocumentoResponse]:
    fecha_desde = (
        datetime.strptime(filtros.fechaDesde, "%Y-%m-%d")
        if filtros.fechaDesde
        else None
    )
    fecha_hasta = (
        datetime.strptime(filtros.fechaHasta, "%Y-%m-%d")
        if filtros.fechaHasta
        else None
    )

    query = select(
        DocumentoElectronico.ID_Pro,
        DocumentoElectronico.ID_Emi,
        DocumentoElectronico.ID_Rec,
        DocumentoElectronico.TipDoc,
        DocumentoElectronico.NumDoc,
        DocumentoElectronico.DocSus,
        DocumentoElectronico.FecDoc,
        DocumentoElectronico.FecAut,
        DocumentoElectronico.EmaNot,
    ).distinct()

    if filtros.id_emi:
        query = query.where(DocumentoElectronico.ID_Emi == filtros.id_emi)
    if filtros.id_rec:
        query = query.where(DocumentoElectronico.ID_Rec == filtros.id_rec)
    if filtros.tipoDocumento:
        query = query.where(DocumentoElectronico.TipDoc == filtros.tipoDocumento)
    if fecha_desde and fecha_hasta:
        query = query.where(
            DocumentoElectronico.FecDoc.between(fecha_desde, fecha_hasta)
        )
    elif fecha_desde:
        query = query.where(DocumentoElectronico.FecDoc >= fecha_desde)
    elif fecha_hasta:
        query = query.where(DocumentoElectronico.FecDoc <= fecha_hasta)

    documentos = session.exec(query).all()

    response = [
        DocumentoResponse(
            id_emi=doc.ID_Emi,
            id_rec=doc.ID_Rec,
            tipo_documento=doc.TipDoc,
            fecha=doc.FecDoc,
            num_doc=doc.NumDoc,
        )
        for doc in documentos
    ]

    return response
