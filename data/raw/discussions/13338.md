## Accepted answer by YuriiMotov (MEMBER)

There were about 10 mistakes in your code:

This code example works:
```

from fastapi import FastAPI
from fastapi.params import Depends
from fastapi.testclient import TestClient
from pydantic import BaseModel
import pytest


class Model(BaseModel):
    val: str

def get_func1():
    return {
        "id": "",
        "outputs": [
            {
                "name": "output of original function 1",
                "value": "original function 1"
            }
        ],
    }

def get_func2():
    return {
        "id": "",
        "outputs": [
            {
                "name": "output of original function 2",
                "value": "original function 2"
            }
        ],
    }

class SampleAppBuilder:
    @classmethod
    def get_app(cls):
        app = FastAPI()

        @app.get("/endpoint")
        async def get_forecast_trip_budget(
                func1=Depends(get_func1),
                func2=Depends(get_func2)
        ):
            """
            Returns the response.
            """
            val = func1["outputs"][0]["value"]
            return Model(
                val = val
            )

        return app


@pytest.fixture(scope="function", name="app")
def get_app():
    """Get a FastAPI app"""
    yield SampleAppBuilder.get_app()



@pytest.fixture(scope="function")
def sample_fastapi_client(app: FastAPI):
    """Get a FastAPI test client for the actual app"""
    yield TestClient(app)

async def mock_func1_function():
    return {
        "id": "",
        "outputs": [
            {
                "name": "output",
                "value": "I am mocked"
            }
        ],
    }

async def mock_func2_function():
    return {
        "id": "",
        "outputs": [
            {
                "name": "output",
                "value": "I am mocked 2"
            }
        ],
    }

async def test_sample_class(
    sample_fastapi_client: TestClient,
    app: FastAPI,
):
    app.dependency_overrides[get_func1] = mock_func1_function
    app.dependency_overrides[get_func2] = mock_func2_function
    response = sample_fastapi_client.get("/endpoint")
    assert response.status_code == 200
    assert response.json()["val"] == "I am mocked"

```
