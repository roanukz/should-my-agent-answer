### Example Code

```python
import pandas as pd
import pathlib
from fastapi import FastAPI, HTTPException

service = FastAPI()
current_path = pathlib.Path(__file__).parents[1]
file_path = current_path / './database/db.csv'

@service.get(path='/user/{password}')
async def read_user(password: int):
    try:
        df = pd.read_csv(file_path, encoding='utf-8')
        user_data = df.query('passwords == @password').to_dict(orient='records')
        if not user_data:
            raise HTTPException(status_code=404, detail=f"Error: User {password} not found")
        return user_data
    except FileNotFoundError:
        raise HTTPException(status_code=404, detail=f"Error: File not found at {file_path}")
    except pd.errors.ParserError:
        raise HTTPException(status_code=400, detail="Error: Failed to parse the CSV file")
```
