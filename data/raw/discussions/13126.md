### Example Code

```python
from fastapi import FastAPI, Request
from fastapi.responses import StreamingResponse

app = FastAPI()

@app.post("/stream-uppercase")
async def stream_uppercase(request: Request):
    """
    Endpoint that reads the request body as a stream,
    extracts only uppercase letters, and streams them back.
    """

    async def uppercase_generator():
        async for chunk in request.stream():
            text = chunk.decode("utf-8", errors="ignore")
            uppercase_only = "".join([ch for ch in text if ch.isupper()])
            if uppercase_only:
                yield uppercase_only

    return StreamingResponse(uppercase_generator(), media_type="text/plain")
```
