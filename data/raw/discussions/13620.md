### Example Code

```python
from typing import Annotated, Callable
from fastapi import Depends, FastAPI, Header

app = FastAPI()

def get_user_browser(user_agent: str = Header(default="")):
    browser = user_agent.split(' ')[0]
    return browser

def get_greeting_message(browser: Annotated[str, Depends(get_user_browser)]):
    return f"Hello, {browser} user!"

class GreetingGenerator:    
    def __init__(self, greeting_provider: Callable[..., str]):
        self.greeting_provider = greeting_provider
    
    def __call__(self):
        message = self.greeting_provider()
        return {"message": message}

greeter = GreetingGenerator(greeting_provider=Depends(get_greeting_message))

@app.get("/hello")
async def hello(
    greeting: Annotated[dict, Depends(greeter)]
):
    return greeting
```


### Description

How to pass a "callback" callable that itself uses dependencies to a dependency?

Using the approach above, it throws a `TypeError: 'Depends' object is not callable`.

## Accepted answer by YuriiMotov (MEMBER)

Elaborated on this a little bit.

The only way I can see (without using third-party dependency injection system) is:

```
from abc import ABC, abstractmethod
from typing import Annotated

from fastapi import Depends, FastAPI, Header

app = FastAPI()


def get_user_browser(user_agent: str = Header(default="")):
    browser = user_agent.split(" ")[0]
    return browser


def get_greeting_message(browser: Annotated[str, Depends(get_user_browser)]):
    return f"Hello, {browser} user!"


class GreetingGenerator(ABC):
    def format_greeting(self, message: str):
        return {"message": message}

    @abstractmethod
    def __call__(self, message: str):
        raise NotImplementedError()


class GreetingGenerator_1(GreetingGenerator):
    def __call__(self, message: str = Depends(get_greeting_message)):
        return self.format_greeting(message)


greeter = GreetingGenerator_1()


@app.get("/hello")
async def hello(greeting: Annotated[dict, Depends(greeter)]):
    return greeting

```

So, you can implement common logic (like response formatting) in `GreetingGenerator`, but you will still need to create new class for every greeting provider
