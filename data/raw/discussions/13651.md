## Accepted answer by sachinh35 (NONE)

I think `BaseHTTPMiddleware` is bypassing the base exception handlers and your `unauthorized_exception_handler` method is not getting called correctly. If you really want to make it work with your way of the code structure, here's how it would work:

```python3
from fastapi import FastAPI, Request
from fastapi import HTTPException
from fastapi.responses import JSONResponse
from starlette.middleware.base import BaseHTTPMiddleware

app = FastAPI()


class UnauthorizedException(HTTPException):
    def __init__(self, detail: str = "未授权的访问"):
        super().__init__(status_code=401, detail=detail)


async def unauthorized_exception_handler(request: Request, exc: UnauthorizedException):
    return JSONResponse(
        status_code=exc.status_code,
        content={"code": exc.status_code, "message": exc.detail},
    )


# 添加异常处理器
app.add_exception_handler(UnauthorizedException, unauthorized_exception_handler)


class auth(BaseHTTPMiddleware):
    def __init__(self, app):
        super().__init__(app)

    async def dispatch(self, request: Request, call_next):
        try:
            raise UnauthorizedException("缺少认证令牌")
        except UnauthorizedException as e:
            return await unauthorized_exception_handler(request, e)


# 添加中间件
app.add_middleware(auth)


@app.get("/protected")
async def protected_route():
    return {"message": "这是一个受保护的路由"}
```

```python3
from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse

app = FastAPI()


@app.middleware("http")
async def auth_middleware(request: Request, call_next):
   # try:
       # your auth check code here...
   # except Exception as e:
       # Some more code here before returning response back
    return JSONResponse(
        status_code=401,
        content={"code": 401, "message": "缺少认证令牌"},
    )


@app.get("/protected")

    return {"message": "这是一个受保护的路由"}
```
