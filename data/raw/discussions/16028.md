### Example Code

```python
from fastapi import FastAPI
from pydantic import BaseModel

app = FastAPI()

# custom error pydantic model that populate "detail" prop of (FastApi) HTTPException
# NOTE: this is a base class that need to be extended
class HttpExpectedError(BaseModel):
  code: str = Field(..., description="Error code like PLAYLIST_NOT_FOUND")
  message: str = Field(..., description="Error message for human")
  def toFastApiJson(self):
    return self.model_dump()

# concret error pydantic model for one route (that extends HttpExpectedError)
class HttpError_PlaylistNotFound(HttpExpectedError):
  code: Literal["PLAYLIST_NOT_FOUND"] = "PLAYLIST_NOT_FOUND"

# route

  responses={
    404: {"model": HttpError_PlaylistNotFound}
  }
)
async def getOne():
  # ...
  if not playlist:
    raise HTTPException(
      status_code=404,
      detail=HttpError_PlaylistNotFound(
        message=f"Playlist {playlist_id} not found"
      ).toFastApiJson()
    )
```


### Description

The code works, but:
- the openapi spec schema (json) generated by FastApi and used by Docs UIs (swagger/scalar...) will make UIs think that JSON response would be:
    ```json
    {
      "code": "PLAYLIST_NOT_FOUND",
      "message": "string"
    }
    ```
- ...while the real returned JSON is:
    ```json
    {
      "detail": {
        "code": "PLAYLIST_NOT_FOUND",
        "message": "string"
      }
    }
    ```

**The problem is the missing `detail` prop in the schema**, how can i add it to the schema generated by fast api?

## Accepted answer by YuriiMotov (MEMBER)

You can make FastAPI return response that follows your schema by adding an exception handler:

```py
from typing import Literal

from fastapi import FastAPI, HTTPException
from fastapi.responses import JSONResponse
from fastapi.testclient import TestClient
from pydantic import BaseModel, Field

app = FastAPI()


@app.exception_handler(404)
async def not_found_handler(request, exc: HTTPException):
    return JSONResponse(
        status_code=404,
        content={"code": exc.detail.get("code"), "message": exc.detail.get("message")},
    )


# custom error pydantic model that populate "detail" prop of (FastApi) HTTPException
# NOTE: this is a base class that need to be extended
class HttpExpectedError(BaseModel):
    code: str = Field(..., description="Error code like PLAYLIST_NOT_FOUND")
    message: str = Field(..., description="Error message for human")

    def toFastApiJson(self):
        return self.model_dump()


# concret error pydantic model for one route (that extends HttpExpectedError)
class HttpError_PlaylistNotFound(HttpExpectedError):
    code: Literal["PLAYLIST_NOT_FOUND"] = "PLAYLIST_NOT_FOUND"


# route
@app.get(
    path="/playlists/{playlist_id}",
    responses={404: {"model": HttpError_PlaylistNotFound}},
)
async def getOne(playlist_id: str):
    playlist = None

    if not playlist:
        raise HTTPException(
            status_code=404,
            detail=HttpError_PlaylistNotFound(
                message=f"Playlist {playlist_id} not found"
            ).toFastApiJson(),
        )


def test_404():
    client = TestClient(app)

    response = client.get("/playlists/123")
    assert response.status_code == 404
    assert response.json() == {
        "code": "PLAYLIST_NOT_FOUND",
        "message": "Playlist 123 not found",
    }
```

Or use generic response wrapper to add `detail` property to the schema in openapi:

```py
class ErrorResponse(BaseModel, Generic[ErrorT]):
    detail: ErrorT


# route
@app.get(
    path="/playlists/{playlist_id}",
    responses={404: {"model": ErrorResponse[HttpError_PlaylistNotFound]}},
)
async def getOne(playlist_id: str):
    playlist = None

    if not playlist:
        raise HTTPException(
            status_code=404,
            detail=HttpError_PlaylistNotFound(
                message=f"Playlist {playlist_id} not found"
            ).toFastApiJson(),
        )

```
