```python
from typing import AsyncGenerator, Optional, Annotated, Any
from datetime import datetime

from fastapi import FastAPI, HTTPException, Depends, Query
from pydantic import BaseModel

from sqlalchemy.ext.asyncio import (
    AsyncSession,
    AsyncAttrs,
    create_async_engine,
    async_sessionmaker,
)
from sqlalchemy.orm import (
    DeclarativeBase,
    relationship,
    mapped_column,
    joinedload,
    load_only,
    Mapped,
)
from sqlalchemy import (
    ForeignKey,
    select,
    func,
    JSON,
)

DATABASE_URL = "sqlite+aiosqlite:///:memory:"


class Base(DeclarativeBase, AsyncAttrs):
    pass


# ORM models
class User(Base):
    __tablename__ = "users"
    id: Mapped[int] = mapped_column(primary_key=True)
    username: Mapped[str] = mapped_column(unique=True, nullable=False)
    password: Mapped[str]


class DBRoom(Base):
    __tablename__ = "rooms"
    id: Mapped[int] = mapped_column(primary_key=True)
    name: Mapped[str] = mapped_column(nullable=False)
    owner_id: Mapped[int] = mapped_column(ForeignKey("users.id"), nullable=False)

    owner: Mapped[User] = relationship()
    messages: Mapped[list["DBMessage"]] = relationship(
        back_populates="room", cascade="all, delete-orphan"
    )


class DBMessage(Base):
    __tablename__ = "messages"
    id: Mapped[int] = mapped_column(primary_key=True)
    room_id: Mapped[int] = mapped_column(ForeignKey("rooms.id"), nullable=False)
    user_id: Mapped[int] = mapped_column(ForeignKey("users.id"), nullable=False)
    content: Mapped[str] = mapped_column(nullable=False)
    attachments: Mapped[Optional[list[str]]] = mapped_column(type_=JSON)
    additional_data: Mapped[str]
    metadatas: Mapped[list[dict]] = mapped_column(type_=JSON)
    created_at: Mapped[datetime] = mapped_column(
        server_default=func.now(),
    )

    room: Mapped[DBRoom] = relationship(back_populates="messages")
    user: Mapped[User] = relationship()


# Pydantic schemas
class UserOut(BaseModel, from_attributes=True):
    id: int
    username: str


class MessageOut(BaseModel, from_attributes=True):
    id: int
    content: Optional[str] = None
    attachments: Optional[list[str]] = None
    additional_data: Optional[str] = None
    metadatas: Optional[list[dict]] = None
    created_at: Optional[datetime] = None

    user: Optional[UserOut] = None


class RoomOut(BaseModel, from_attributes=True):
    id: int
    name: str
    owner: UserOut


app = FastAPI()

# Async engine and session
engine = create_async_engine(DATABASE_URL, echo=False, future=True)
AsyncSessionLocal = async_sessionmaker(
    engine, expire_on_commit=False, class_=AsyncSession
)


async def get_session() -> AsyncGenerator[AsyncSession, None]:
    async with AsyncSessionLocal() as session:
        yield session


# Startup: create tables and seed basic data
@app.on_event("startup")
async def on_startup():
    async with engine.begin() as conn:
        await conn.run_sync(Base.metadata.create_all)

    # seed minimal data if none
    async with AsyncSessionLocal() as session:
        res = await session.execute(select(User))
        users = res.scalars().all()
        if not users:
            alice = User(username="alice", password="1234")
            bob = User(username="bob", password="5678")
            session.add_all([alice, bob])
            await session.commit()

            # create rooms
            r1 = DBRoom(name="General", owner_id=alice.id)
            r2 = DBRoom(name="Random", owner_id=bob.id)
            session.add_all([r1, r2])
            await session.commit()

            # add messages
            m1 = DBMessage(
                room_id=r1.id,
                user_id=alice.id,
                content="Welcome to General!",
                additional_data="test",
                metadatas=[{"a": 1, "b": 2}],
            )
            m2 = DBMessage(
                room_id=r1.id,
                user_id=bob.id,
                content="Hi everyone",
                attachments=["file1.txt"],
                additional_data="from 123",
                metadatas=[{"a": 1, "b": 3}],
            )
            m3 = DBMessage(
                room_id=r2.id,
                user_id=bob.id,
                content="Random thoughts here",
                additional_data="from service b",
                metadatas=[{"a": 1, "b": 2}],
            )
            session.add_all([m1, m2, m3])
            await session.commit()


@app.get("/rooms", response_model=list[RoomOut])
async def list_rooms(*, session: AsyncSession = Depends(get_session)):
    result = await session.execute(select(DBRoom).options(joinedload(DBRoom.owner)))
    rooms = result.scalars().all()
    return rooms


@app.get("/rooms/{room_id}/messages", response_model=list[MessageOut])
async def get_messages(
    room_id: int,
    columns: Annotated[
        Optional[list[str]], Query(description="Columns to include")
    ] = None,
    *,
    session: AsyncSession = Depends(get_session)
):
    """Gets all messages for a specific chat room with pagination. Use '*' in columns to get all columns."""
    if columns is not None and "*" in columns:
        # select all
        query = (
            select(DBMessage)
            .options(joinedload(DBMessage.user))
            .options(joinedload(DBMessage.room))
            .where(DBMessage.room_id == room_id)
        )
    else:
        select_columns = [
            DBMessage.id,
            DBMessage.user_id,
            DBMessage.content,
            DBMessage.attachments,
            DBMessage.additional_data,
            DBMessage.metadatas,
            DBMessage.created_at,
        ]
        load_user = True
        load_chat_room = False
        if columns:
            if "id" not in columns:
                columns.insert(0, "id")
            if "user" in columns:
                columns.remove("user")
            else:
                load_user = False
            if "room" in columns:
                load_chat_room = True
                columns.remove("room")
            else:
                load_chat_room = False
            # only select specific columns
            select_columns = [
                getattr(DBMessage, col) for col in columns if hasattr(DBMessage, col)
            ]

        # default reply
        # select all
        query = (
            select(
                DBMessage,
            )
            .where(DBMessage.room_id == room_id)
            .options(load_only(*select_columns))
        )
        if load_user:
            query = query.options(joinedload(DBMessage.user))
        if load_chat_room:
            query = query.options(joinedload(DBMessage.room))

    result = await session.execute(query)
    messages = result.scalars().all()
    return messages


if __name__ == "__main__":
    import uvicorn

    uvicorn.run(app, host="127.0.0.1", port=8000)
```

I need to keep the function signature as is because this is used in other places such as batch process messages. Adding `response_model_exclude_unset=True` doesn't work on the `chat/{id}/messages` path as during Pydantic serialisation the parameters will still be accessed. 

Removing `response_model` and `response_model_exclude_unset` will work, but the user password will then be returned in the response. Changing `query.options(joinedload(DBMessage.user))` to `query.options(joinedload(DBMessage.user).load_only(User.id, User.username))` doesn't remove the password field without `response_model`.

## Accepted answer by NightFurySL2001 (NONE)

I think a better way might be to introduce response output functions specific for routing. It is actually quite common to write a function that returns  some Python-specific stuff, and then another function that converts the Python-specific stuff to a HTTP response object (or JSON-encodable).

For example, a module utilising pandas might do this:
```py
def process_df(columns: list[str], ...) -> DataFrame:
    df = DataFrame()
    return df

@router.get("/example")
def process_df_to_json(columns: list[str], ...):
    df = process_df(columns, ...)
    return df.to_dict()
```

In this example `process_df_to_json` had to repeat the same method signature as `process_df`, however this function is useless inside Python. Introducing a response output function might help to clearly separate business functions (e.g. `process_df`) with presentation funtions (e.g. `process_df_to_json`).
