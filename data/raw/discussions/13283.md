Can anyone explain me why is that ?
My expectation was to see something similar to this: [Testing WebSockets](https://fastapi.tiangolo.com/advanced/testing-websockets/#testing-websockets)

## Accepted answer by YuriiMotov (MEMBER)

It depends on whether you interact with server how it expects or not.

```
from fastapi import FastAPI, WebSocketDisconnect
from fastapi.testclient import TestClient
from fastapi.websockets import WebSocket
import pytest

app = FastAPI()

@app.websocket("/ws")
async def websocket(websocket: WebSocket):
    # Send data and close connection
    await websocket.accept()
    await websocket.send_json({"msg": "Hello WebSocket"})
    await websocket.close()

# ====  Tests  ====

def test_websocket():
    # Connect to server, receive data from server, server closes connection
    client = TestClient(app)
    with client.websocket_connect("/ws") as websocket:
        data = websocket.receive_json()
        assert data == {"msg": "Hello WebSocket"}
```

In the code example above everything is as server expects: client connected, server sends data to client and closes connection (exit endpoint function).

In the next code example, server expects data from user.
```
from fastapi import FastAPI, WebSocketDisconnect
from fastapi.testclient import TestClient
from fastapi.websockets import WebSocket
import pytest

app = FastAPI()

@app.websocket("/ws2")
async def websocket2(websocket: WebSocket):
    # Wait for data from the client and then close connection
    await websocket.accept()
    await websocket.receive_json()  # Wait for data from client
    await websocket.close()

# ====  Tests  ====

def test_websocket2():
    # Connect to server, send data to server, server recieves data and closes connection
    client = TestClient(app)
    with client.websocket_connect("/ws2") as websocket:
        websocket.send_json({"msg": "Hello WebSocket"})


def test_websocket2_clientside_close():
    # Connect to server, server is waiting for data, but we just close connection
    client = TestClient(app)
    with pytest.raises(WebSocketDisconnect):
        with client.websocket_connect("/ws2") as _websocket:
            pass  # Do not send data, just close connection
```

If we do send data (`test_websocket2`), server receives data and closes connection.
But if we don't send data and just close connection (`test_websocket2_clientside_close`), `WebSocketDisconnect` exception is raised inside the endpoint.


Usually we handle this exception inside the endpoint function:
```
from fastapi import FastAPI, WebSocketDisconnect
from fastapi.testclient import TestClient
from fastapi.websockets import WebSocket

app = FastAPI()


@app.websocket("/ws2")
async def websocket2(websocket: WebSocket):
    # Wait for data from the client and then close connection
    await websocket.accept()
    try:
        await websocket.receive_json()
        await websocket.close()
    except WebSocketDisconnect:
        pass

# ====  Tests  ====

def test_websocket2_clientside_close():
    # Connect to server, server is waiting for data, but we just close connection
    client = TestClient(app, raise_server_exceptions=False)
    with client.websocket_connect("/ws2") as _websocket:
        pass  # Do not send data, just close connection
```
